import {Meta, ColorPalette, ColorItem} from '@storybook/addon-docs/blocks';
import {stops, lightTheme, darkTheme} from '../../styles/color-definitions.js';

<Meta title="Colors" />

## Light Theme

<ColorPalette>
  {lightTheme.contrastColors
    .filter((scale) => scale.values)
    .map((scale) => (
      <ColorItem
        key={scale.name}
        title={scale.name}
        colors={Object.fromEntries(
          scale.values.map((val, i) => [`${stops[i]}`, val.value])
        )}
      />
    ))}
</ColorPalette>
